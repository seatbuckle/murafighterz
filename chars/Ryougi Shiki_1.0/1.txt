; ============================================================
; ユーザ設定ファイルです。(注：★の箇所は書き換えないで下さい)
; ============================================================
; ここでは、特殊ゲージであるMCゲージ / ガードゲージ / ヴァイタルゲージ
; の3つのゲージ位置を変更することができます。
; 画面上におけるX軸とY軸の数値を入力して下さい。
; 尚、ガードゲージとヴァイタルゲージは連動して動きます。
;
;
; 〓〓〓〓〓〓〓〓
; 書き換え方法
; 〓〓〓〓〓〓〓〓
; 以下にMCゲージとガードゲージのXY軸の数値を入力する箇所が
; あります。mugenの画面上における、このX軸とY軸の数値を変える
; ことによってそれぞれのゲージの位置を変更します。
;
; 尚、mugenの画面はY軸では上方向がマイナス、X軸では右方向が
; プラスとなります。
; ┏━━━━━━━━━━━━┓
; ┃                        ┃↑
; ┃                        ┃Y
; ┃                        ┃軸
; ┃       mugen画面        ┃マ
; ┃                        ┃イ
; ┃                        ┃ナ
; ┃                        ┃ス
; ┃                        ┃側
; ┃                        ┃
; ┗━━━━━━━━━━━━┛
;        X軸プラス側→
;
;
; 【例１】MCゲージを23ドット右に動かす場合
;
; ①.MCゲージのX軸の数値を決めている箇所を探す(この場合は以下になる)
; [State 5900, MCゲージ座標　X軸]　←どの座標かはここで確認できるお
; type = VarSet
; trigger1 = 1
; var(52) = 0		←デフォルトでは0になっている
; persistent = 0
;
; ②."var(52) = 0" という箇所の最後の数値(デフォルトでは0)を変える。
; ここの例では23ドットゲージ位置を右に動かすため、"23"と入力する(左に動かす場合は-23)。
;
; [State 5900, MCゲージ座標　X軸]
; type = VarSet
; trigger1 = 1
; var(52) = 23		←変えた
; persistent = 0
;
; ※"var(52) = 0" の最後の数値以外は変更する必要なし
;
;------------------------------------------------------------
; 【例２】ガードゲージを100ドット上に動かす場合
;
; ①.ガードゲージのY軸の数値を決めている箇所を探す(この場合は以下になる)
;[State 5900, ガードゲージ座標　Y軸]
;type = VarSet
;trigger1 = 1
;var(55) = 0		←デフォルトでは0になっている
;persistent = 0
;
; ②."var(55) = 0" という箇所の最後の数値(デフォルトでは0)を変える。
; ここの例では100ドットゲージ位置を上に動かすため、"-100"と入力する(下に動かす場合は100)。
;
;[State 5900, ガードゲージ座標　Y軸]
;type = VarSet
;trigger1 = 1
;var(55) = -100		←変えた
;persistent = 0
;
; ※"var(55) = 0" の最後の数値以外は変更する必要なし
;
;
; 【テンプレ】
;
; jojoのスタンドゲージと同じ位置にガードゲージを表示
; --------------------------------------------------
; ガードゲージX軸の値：-65
; ガードゲージY軸の値：-180
;
;
;
; 上記の要領で、下記を変更して下さい。
;
